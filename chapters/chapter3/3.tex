\section{Trzeci algorytm (\textbf{Metoda19})}
\textbf{Metoda19} jest wynikiem rozwoju \textbf{metody5}, która opierała się na łączeniu wyników różnych algorytmów. W \textbf{metodzie5} zastosowano \textbf{losowy las} i \textbf{XGBoost} jako modele bazowe, które następnie były integrowane za pomocą \textbf{regresji logistycznej}. \textbf{Metoda19} rozwija tę koncepcję, wykorzystując bardziej rozbudowaną architekturę \textbf{warstwowego łączenia modeli} z trzema różnymi modelami bazowymi: \textbf{losowym lasem}, \textbf{XGBoostem} oraz \textbf{wzmacnianiem gradientowym (Gradient Boosting)}.

W \textbf{metodzie19} łączone są \textbf{losowy las}, oparty na agregacji wielu losowych predyktorów, \textbf{XGBoost} oraz \textbf{wzmacnianie gradientowe (Gradient Boosting)}, które w ustalonej kolejności buduje modele korygujące błędy poprzedników. Inspiracją dla \textbf{metody19} jest zastosowanie \textbf{regresji logistycznej} jako meta-modelu, który skutecznie integruje predykcje tych trzech algorytmów. Wykorzystanie różnorodnych modeli bazowych ma na celu zwiększenie ogólnej dokładności i stabilności predykcji.

\subsection{Wejściowe dane}
Algorytm przyjmuje następujące parametry wejściowe:

\begin{enumerate}
	\def\labelenumi{\arabic{enumi}.}
	\item
	      \textbf{X\_train}: macierz danych treningowych dla n próbek, m cech. Każdy wiersz reprezentuje próbkę, a każda kolumna cechę. Wartości w macierzy są zwykle liczbowe.
	\item
	      \textbf{y\_train}: wektor etykiet dla danych treningowych. Rozmiar: n, gdzie n - liczba próbek w danych treningowych.
	\item
	      \textbf{X\_test}: macierz danych testowych, analogiczna do \textbf{X\_train}, używana do oceny modelu.
	\item
	      \textbf{y\_test}: wektor etykiet testowych, odpowiadający danym testowym.
\end{enumerate}

\subsection{Wyjściowe dane}
Algorytm zwraca następujące elementy:

\begin{enumerate}
	\def\labelenumi{\arabic{enumi}.}
	\item
	      \textbf{meta\_model}: meta-model (\textbf{regresja logistyczna}) wytrenowany na danych pochodzących z predykcji modeli \textbf{losowy las}, \textbf{XGBoost} oraz \textbf{wzmacniania gradientowego}.
	\item
	      \textbf{meta\_features\_test}: macierz meta-cech, zbudowana na podstawie predykcji modeli bazowych dla danych testowych.
	\item
	      \textbf{y\_test}: wektor etykiet testowych, identyczny z wejściowym \textbf{y\_test}.
\end{enumerate}

\subsection{Algorytm}
\begin{enumerate}
	\def\labelenumi{\arabic{enumi}.}
	\item
	      Skalowanie danych: Przeprowadza się przeskalowanie \textbf{X\_train} i \textbf{X\_test} za pomocą \textbf{StandardScaler}.
	\item
	      Inicjalizacja i optymalizacja modeli bazowych:
	      \begin{itemize}
		      \item Inicjalizuje się \textbf{losowy las} skonfigurowany z 200 estymatorami, maksymalną głębokością drzew równą 10 oraz kontrolowaną losowością (\textbf{random\_state=42}).
		      \item Inicjalizuje się \textbf{XGBClassifier} z ustawionymi 200 estymatorami, wyłączonym automatycznym kodowaniem etykiet, metryką oceny log-loss, tempem uczenia 0.05, głębokością drzew ograniczoną do 5 oraz ziarnem losowości (\textbf{random\_state=42}).
		      \item Inicjalizuje się \textbf{wzmacnianie gradientowe (GradientBoostingClassifier)} z 100 estymatorami, tempem uczenia 0.1, maksymalną głębokością drzew równą 3 oraz losowością (\textbf{random\_state=42}).
	      \end{itemize}
	\item
	      Generowanie meta-cech za pomocą walidacji krzyżowej z 5 podziałami:
	      \begin{itemize}
		      \item Uzyskuje się probabilistyczne predykcje dla klasy pozytywnej z każdego modelu bazowego na \textbf{X\_train\_scaled} za pomocą \textbf{cross\_val\_predict}.
		      \item Tworzy się macierz meta-cech \textbf{meta\_features\_train} z predykcji \textbf{losowego lasu}, \textbf{XGBoost} oraz \textbf{wzmacniania gradientowego}.
	      \end{itemize}
	\item
	      Trenowanie meta-modelu:
	      \begin{itemize}
		      \item Inicjalizuje się \textbf{regresję logistyczną (\textbf{LogisticRegression})} z \textbf{solver='liblinear'} i \textbf{random\_state=42}.
		      \item Przeprowadza się trening meta-modelu na \textbf{meta\_features\_train} i \textbf{y\_train}.
	      \end{itemize}
	\item Generowanie predykcji na zbiorze testowym:
	      \begin{itemize}
		      \item Przeprowadza się trening każdego modelu bazowego na całym \textbf{X\_train\_scaled}.
		      \item Uzyskuje się probabilistyczne predykcje dla klasy pozytywnej z każdego wytrenowanego modelu na \textbf{X\_test\_scaled}.
		      \item Tworzy się macierz meta-cech \textbf{meta\_features\_test} z predykcji \textbf{losowego lasu}, \textbf{XGBoost} oraz \textbf{wzmacniania gradientowego} dla danych testowego.
	      \end{itemize}
	\item Zwracanie wyników: zwracany jest wytrenowany \textbf{meta\_model}, \textbf{meta\_features\_test} oraz \textbf{y\_test}.
\end{enumerate}

\subsection{Właściwości analityczne}
\begin{enumerate}
	\def\labelenumi{\arabic{enumi}.}
	\item
	      \textbf{Efektywność}:
	      \begin{itemize}
		      \item Wykorzystanie różnorodnych algorytmów predykcyjnych, takich jak oparty na zespołach predyktorów \textbf{losowy las} oraz techniki \textbf{wzmacnianie gradientowe} i \textbf{XGBoost}, umożliwia uchwycenie złożonych zależności w danych.
		      \item Zastosowanie \textbf{walidacji krzyżowej} do generowania meta-cech minimalizuje ryzyko przeuczenia.
		      \item Meta-model (\textbf{regresja logistyczna}) efektywnie łączy predykcje z różnych modeli bazowych.
	      \end{itemize}
	\item
	      \textbf{Stabilność}:
	      \begin{itemize}
		      \item Modele bazowe: \textbf{losowy las}, \textbf{wzmacnianie gradientowe} i \textbf{XGBoost} są stosunkowo stabilne dzięki swoim mechanizmom agregacji i korekcji błędów.
		      \item Warstwowe łączenie modeli z walidacją krzyżową pomaga uzyskać bardziej stabilne predykcje meta-modelu.
	      \end{itemize}
	\item
	      \textbf{Złożoność obliczeniowa}:
	      \begin{itemize}
		      \item Trening wielu modeli bazowych i meta-modelu zwiększa złożoność obliczeniową w porównaniu z pojedynczymi modelami.
		      \item Walidacja krzyżowa dodatkowo wydłuża czas treningu.
	      \end{itemize}
	\item
	      \textbf{Wydajność}: łączenie różnorodnych stron algorytmów często prowadzi do wyższej dokładności predykcji niż pojedyncze modele.
\end{enumerate}
