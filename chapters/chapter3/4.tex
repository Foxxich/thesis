\section{Czwarty algorytm (\textbf{Metoda20})}
\textbf{Metoda20} powstała na bazie idei integracji wielu modeli, gdzie różne klasyfikatory były łączone w celu poprawy ogólnej wydajności predykcyjnej. Inspiracją były metody \textbf{warstwowego łączenia modeli}, w których predykcje modeli bazowych służą jako cechy dla meta-modelu.

W \textbf{Metodzie20} połączono \textbf{LightGBM}, \textbf{CatBoost}, \textbf{losowy las} oraz \textbf{wzmacnianie gradientowe}. \textbf{LightGBM} i \textbf{CatBoost} to zaawansowane algorytmy uczenia wzmacniającego, znane ze swojej szybkości i efektywności. \textbf{Regresja logistyczna} służy jako meta-model, integrując probabilistyczne predykcje tych czterech różnorodnych modeli bazowych. Celem jest wykorzystanie mocnych stron każdego z algorytmów, aby osiągnąć wyższą \textbf{dokładność} przy mniejszym \textbf{czasie wykonania}.

\subsection{Wejściowe dane}
Metoda \textbf{train\_run20} przyjmuje następujące parametry wejściowe:

\begin{enumerate}
	\def\labelenumi{\arabic{enumi}.}
	\item
	      \textbf{X\_train}: Macierz danych treningowych dla n próbek, m cech. Każdy wiersz reprezentuje próbkę, a każda kolumna cechę. Wartości w macierzy są zwykle liczbowe.
	\item
	      \textbf{y\_train}: Wektor etykiet dla danych treningowych. Rozmiar: n, gdzie n - liczba próbek w danych treningowych.
	\item
	      \textbf{X\_test}: Macierz danych testowych, analogiczna do \textbf{X\_train}, używana do oceny modelu.
	\item
	      \textbf{y\_test}: Wektor etykiet testowych, odpowiadający danym testowym.
\end{enumerate}

\subsection{Wyjściowe dane}
Metoda zwraca następujące elementy:

\begin{enumerate}
	\def\labelenumi{\arabic{enumi}.}
	\item
	      \textbf{meta\_model}: Meta-model (\textbf{regresja logistyczna}) wytrenowany na predykcjach \textbf{LightGBM}, \textbf{CatBoost}, \textbf{losowego lasu} oraz \textbf{wzmacniania gradientowego}.
	\item
	      \textbf{meta\_features\_test}: Macierz meta-cech (predykcje modeli bazowych) dla danych testowych.
	\item
	      \textbf{y\_test}: Wektor etykiet testowych.
\end{enumerate}

\subsection{Algorytm}
\begin{enumerate}
	\def\labelenumi{\arabic{enumi}.}
	\item
	      Trenowanie modeli bazowych:
	      \begin{itemize}
		      \item Przeprowadza się trening \textbf{LGBMClassifier} na \textbf{X\_train} i \textbf{y\_train}.
		      \item Przeprowadza się trening \textbf{CatBoostClassifier} na \textbf{X\_train} i \textbf{y\_train}.
		      \item Przeprowadza się trening \textbf{klasyfikatora losowego lasu (\texttt{RandomForestClassifier})} na \textbf{X\_train} i \textbf{y\_train}.
		      \item Przeprowadza się trening \textbf{klasyfikatora wzmacniania gradientowego (\texttt{GradientBoostingClassifier})} na \textbf{X\_train} i \textbf{y\_train}.
	      \end{itemize}
	\item
	      Generowanie predykcji dla danych treningowych:
	      \begin{itemize}
		      \item Uzyskuje się probabilistyczne predykcje klasy pozytywnej z każdego modelu bazowego na \textbf{X\_train}.
		      \item Tworzy się macierz meta-cech \textbf{meta\_features\_train}, gdzie każda kolumna zawiera predykcje jednego z modeli bazowych.
	      \end{itemize}
	\item
	      Trenowanie meta-modelu:
	      \begin{itemize}
		      \item Przeprowadza się trening \textbf{regresji logistycznej} na \textbf{meta\_features\_train} i \textbf{y\_train}.
	      \end{itemize}
	\item
	      Generowanie predykcji dla danych testowych:
	      \begin{itemize}
		      \item Uzyskuje się probabilistyczne predykcje klasy pozytywnej z każdego wytrenowanego modelu bazowego na \textbf{X\_test}.
		      \item Tworzy się macierz meta-cech \textbf{meta\_features\_test}, analogicznie do macierzy treningowej.
	      \end{itemize}
	\item
	      Zwracanie wyników: Zwracany jest wytrenowany \textbf{meta\_model}, \textbf{meta\_features\_test} oraz \textbf{y\_test}.
\end{enumerate}

\subsection{Właściwości analityczne}
\begin{enumerate}
	\def\labelenumi{\arabic{enumi}.}
	\item
	      \textbf{Efektywność}: Łączenie predykcji wielu modeli ma na celu zwiększenie \textbf{dokładności predykcji}.
	\item
	      \textbf{Wykorzystanie różnorodności modeli}: Zastosowanie różnych typów algorytmów pozwala na uwzględnienie różnych aspektów danych i wzorców.
	\item
	      \textbf{Meta-model}: \textbf{Regresja logistyczna} skutecznie integruje predykcje modeli bazowych, ucząc się, jak najlepiej je łączyć.
	\item
	      \textbf{Złożoność obliczeniowa}: Trening wielu złożonych modeli może być kosztowny obliczeniowo, ale predykcja jest stosunkowo szybka.
	\item
	      \textbf{Wydajność}: Oczekuje się, że kombinacja tych technik modelowania zapewni wysoką wydajność w zadaniu klasyfikacji.
\end{enumerate}
